cmake_minimum_required(VERSION 3.20)
project(soundcurrent-eq VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets)
if(WIN32)
  enable_language(RC)
  add_executable(soundcurrent-eq WIN32 src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc data/localization/resources.qrc data/windows.rc)
  target_include_directories(soundcurrent-eq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/data)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-windows-audio Qt6::Widgets Qt6::Network)
else()
  add_executable(soundcurrent-eq src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc data/localization/resources.qrc)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-dsp Qt6::Widgets Qt6::Network)
endif()
if(MSVC)
  target_compile_options(soundcurrent-eq PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-eq PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(soundcurrent-dsp STATIC src/dsp.cpp src/enhancement.cpp)
target_include_directories(soundcurrent-dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(soundcurrent-dsp PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(soundcurrent-dsp-test tests/dsp_response.cpp)
target_link_libraries(soundcurrent-dsp-test PRIVATE soundcurrent-dsp)
if(WIN32 AND MINGW)
  target_link_options(soundcurrent-dsp-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
enable_testing()
if(TARGET Qt6::Core)
  add_executable(soundcurrent-startup-test tests/startup_registration.cpp)
  target_include_directories(soundcurrent-startup-test PRIVATE src)
  target_link_libraries(soundcurrent-startup-test PRIVATE Qt6::Core)
  add_test(NAME startup-registration COMMAND soundcurrent-startup-test)
endif()

add_executable(soundcurrent-endpoint-identity-test tests/windows_endpoint_identity.cpp)
add_test(NAME endpoint-identity COMMAND soundcurrent-endpoint-identity-test)
find_package(Threads REQUIRED)
add_executable(soundcurrent-mic-transport-test tests/windows_mic_transport.cpp)
target_link_libraries(soundcurrent-mic-transport-test PRIVATE Threads::Threads)
add_test(NAME microphone-transport COMMAND soundcurrent-mic-transport-test)
set_tests_properties(microphone-transport PROPERTIES TIMEOUT 20)
add_test(NAME dsp-response COMMAND soundcurrent-dsp-test)
add_test(NAME shared-ui COMMAND soundcurrent-eq --ui-self-test)
set_tests_properties(shared-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)

if(WIN32)
  add_subdirectory(native/windows/driver-manager)
  # Opt-in cross-user Windows mutex test, with no Qt/User32 startup dependency.
  add_executable(soundcurrent-windows-session-gate-smoke tests/windows_session_gate_smoke.cpp)
  target_include_directories(soundcurrent-windows-session-gate-smoke PRIVATE src)
  target_link_libraries(soundcurrent-windows-session-gate-smoke PRIVATE advapi32)
  set_target_properties(soundcurrent-windows-session-gate-smoke PROPERTIES AUTOMOC OFF)
  if(MSVC)
    set_target_properties(soundcurrent-windows-session-gate-smoke PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
  endif()

  add_library(soundcurrent-windows-audio STATIC src/windows_audio.cpp src/windows_io.cpp)
  target_include_directories(soundcurrent-windows-audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(soundcurrent-windows-audio PUBLIC soundcurrent-dsp ole32 uuid avrt advapi32)
  if(MINGW)
    target_link_libraries(soundcurrent-windows-audio PUBLIC ksguid)
  endif()
  add_executable(soundcurrent-route-guardian src/windows_route_guardian.cpp)
  target_link_libraries(soundcurrent-route-guardian PRIVATE soundcurrent-windows-audio shell32)
  if(MINGW)
    target_link_options(soundcurrent-route-guardian PRIVATE -static)
  endif()
  install(TARGETS soundcurrent-route-guardian RUNTIME DESTINATION bin)
  add_executable(soundcurrent-cable-setup-guard src/windows_cable_setup_guard.cpp)
  target_link_libraries(soundcurrent-cable-setup-guard PRIVATE soundcurrent-windows-audio Qt6::Core shell32)
  install(TARGETS soundcurrent-cable-setup-guard RUNTIME DESTINATION bin)
  add_executable(soundcurrent-windows-route-contract tests/windows_route_contract.cpp)
  target_link_libraries(soundcurrent-windows-route-contract PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-route-contract PRIVATE -static-libstdc++ -static-libgcc)
  endif()
  add_test(NAME windows-route-contract COMMAND soundcurrent-windows-route-contract)
  # Opt-in test changes only the clone's audio routes. It kills its own child
  # to check the same managed lease and guardian used by the desktop app.
  add_executable(soundcurrent-windows-route-smoke tests/windows_route_smoke.cpp)
  target_link_libraries(soundcurrent-windows-route-smoke PRIVATE soundcurrent-windows-audio Qt6::Core)
  add_dependencies(soundcurrent-windows-route-smoke soundcurrent-route-guardian)
  # Opt-in guest endpoint volume check; does not generate a test signal.
  add_executable(soundcurrent-windows-volume-smoke tests/windows_volume_route_smoke.cpp)
  target_link_libraries(soundcurrent-windows-volume-smoke PRIVATE soundcurrent-windows-audio Qt6::Core)
  add_dependencies(soundcurrent-windows-volume-smoke soundcurrent-route-guardian)

  # This integration check plays audio only with --run; keep it out of CTest.
  add_executable(soundcurrent-windows-audio-smoke tests/windows_audio_smoke.cpp)
  target_link_libraries(soundcurrent-windows-audio-smoke PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-audio-smoke PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
