cmake_minimum_required(VERSION 3.20)
project(soundcurrent-eq VERSION 0.6.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  set(CMAKE_AUTOMOC ON)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(soundcurrent-eq WIN32 src/main_windows.cpp data/windows.rc)
  target_include_directories(soundcurrent-eq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/data)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-windows-audio comctl32 shell32)
  if(MINGW)
    target_link_options(soundcurrent-eq PRIVATE -municode -static-libstdc++ -static-libgcc)
  endif()
else()
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets)
  add_executable(soundcurrent-eq src/main.cpp)
  target_link_libraries(soundcurrent-eq PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
endif()
if(MSVC)
  target_compile_options(soundcurrent-eq PRIVATE /W4 /permissive-)
else()
  target_compile_options(soundcurrent-eq PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(soundcurrent-dsp STATIC src/dsp.cpp)
target_include_directories(soundcurrent-dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(soundcurrent-dsp PRIVATE /W4 /permissive-)
else()
  target_compile_options(soundcurrent-dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(soundcurrent-dsp-test tests/dsp_response.cpp)
target_link_libraries(soundcurrent-dsp-test PRIVATE soundcurrent-dsp)
if(WIN32 AND MINGW)
  target_link_options(soundcurrent-dsp-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
enable_testing()
add_test(NAME dsp-response COMMAND soundcurrent-dsp-test)

if(WIN32)
  add_library(soundcurrent-windows-audio STATIC src/windows_audio.cpp)
  target_include_directories(soundcurrent-windows-audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(soundcurrent-windows-audio PUBLIC soundcurrent-dsp ole32 uuid ksguid avrt)
  # This integration check plays audio only with --run; keep it out of CTest.
  add_executable(soundcurrent-windows-audio-smoke tests/windows_audio_smoke.cpp)
  target_link_libraries(soundcurrent-windows-audio-smoke PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-audio-smoke PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS soundcurrent-eq RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT WIN32)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES COPYRIGHT DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq RENAME copyright)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
