cmake_minimum_required(VERSION 3.20)
project(soundcurrent-eq VERSION 0.7.3 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets)
if(WIN32)
  enable_language(RC)
  add_executable(soundcurrent-eq WIN32 src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc data/windows.rc)
  target_include_directories(soundcurrent-eq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/data)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-windows-audio Qt6::Widgets Qt6::Network)
else()
  add_executable(soundcurrent-eq src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-dsp Qt6::Widgets Qt6::Network)
endif()
if(MSVC)
  target_compile_options(soundcurrent-eq PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-eq PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(soundcurrent-dsp STATIC src/dsp.cpp src/enhancement.cpp)
target_include_directories(soundcurrent-dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(soundcurrent-dsp PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(soundcurrent-dsp-test tests/dsp_response.cpp)
target_link_libraries(soundcurrent-dsp-test PRIVATE soundcurrent-dsp)
if(WIN32 AND MINGW)
  target_link_options(soundcurrent-dsp-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
enable_testing()
add_test(NAME dsp-response COMMAND soundcurrent-dsp-test)
add_test(NAME shared-ui COMMAND soundcurrent-eq --ui-self-test)
set_tests_properties(shared-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)

if(WIN32)
  add_library(soundcurrent-windows-audio STATIC src/windows_audio.cpp src/windows_io.cpp)
  target_include_directories(soundcurrent-windows-audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(soundcurrent-windows-audio PUBLIC soundcurrent-dsp ole32 uuid avrt)
  if(MINGW)
    target_link_libraries(soundcurrent-windows-audio PUBLIC ksguid)
  endif()
  # This integration check plays audio only with --run; keep it out of CTest.
  add_executable(soundcurrent-windows-audio-smoke tests/windows_audio_smoke.cpp)
  target_link_libraries(soundcurrent-windows-audio-smoke PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-audio-smoke PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS soundcurrent-eq RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT WIN32)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES COPYRIGHT DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq RENAME copyright)
install(FILES THIRD-PARTY-NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
install(DIRECTORY data/speakers DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)

add_executable(soundcurrent-equipment-test tests/equipment_profiles.cpp src/equipment_profiles.cpp data/resources.qrc)
target_include_directories(soundcurrent-equipment-test PRIVATE src)
target_link_libraries(soundcurrent-equipment-test PRIVATE soundcurrent-dsp Qt6::Widgets)
add_test(NAME equipment-profiles COMMAND soundcurrent-equipment-test --ui-self-test)
set_tests_properties(equipment-profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
install(DIRECTORY data/equipment DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)

if(MSVC)
  target_compile_options(soundcurrent-equipment-test PRIVATE /utf-8)
endif()

install(FILES docs/equipment-profiles.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)

add_executable(soundcurrent-processing-guard-test tests/processing_guard.cpp src/processing_guard.cpp)
target_include_directories(soundcurrent-processing-guard-test PRIVATE src)
target_link_libraries(soundcurrent-processing-guard-test PRIVATE Qt6::Core)
add_test(NAME processing-guard COMMAND soundcurrent-processing-guard-test)
set_tests_properties(processing-guard PROPERTIES TIMEOUT 90)
if(MSVC)
  target_compile_options(soundcurrent-processing-guard-test PRIVATE /utf-8)
endif()

if(NOT WIN32)
  add_library(soundcurrent-enhancements MODULE src/enhancement_ladspa.cpp src/enhancement.cpp)
  set_target_properties(soundcurrent-enhancements PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/effects")
  target_compile_definitions(soundcurrent-eq PRIVATE SOUNDCURRENT_EFFECTS_DIR="${CMAKE_INSTALL_FULL_LIBDIR}/soundcurrent-eq")
  install(TARGETS soundcurrent-enhancements LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/soundcurrent-eq)
  install(FILES src/vendor/LGPL-2.1 DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
endif()

add_executable(soundcurrent-enhancement-test tests/enhancement_response.cpp)
target_link_libraries(soundcurrent-enhancement-test PRIVATE soundcurrent-dsp)
add_test(NAME sound-enhancements COMMAND soundcurrent-enhancement-test)
set_tests_properties(sound-enhancements PROPERTIES TIMEOUT 45)
