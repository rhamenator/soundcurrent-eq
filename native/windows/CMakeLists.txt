cmake_minimum_required(VERSION 3.20)
project(soundcurrent-native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(soundcurrent-native-processor STATIC processor.cpp ../../src/dsp.cpp ../../src/enhancement.cpp)
target_include_directories(soundcurrent-native-processor PUBLIC . ../../src)
target_link_libraries(soundcurrent-native-processor PUBLIC Threads::Threads)
add_executable(soundcurrent-native-processor-test processor_test.cpp)
target_link_libraries(soundcurrent-native-processor-test PRIVATE soundcurrent-native-processor)
enable_testing()
add_test(NAME native-processor COMMAND soundcurrent-native-processor-test)
if(WIN32)
  add_library(soundcurrent-native-apo SHARED apo.cpp apo.def)
  target_compile_definitions(soundcurrent-native-apo PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(soundcurrent-native-apo PRIVATE soundcurrent-native-processor ole32 uuid)
  add_executable(soundcurrent-native-endpoint-audit endpoint_audit.cpp)
  target_compile_definitions(soundcurrent-native-endpoint-audit PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(soundcurrent-native-endpoint-audit PRIVATE ole32 uuid advapi32 wintrust)
  add_executable(soundcurrent-native-apo-test apo_test.cpp)
  target_compile_definitions(soundcurrent-native-apo-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(soundcurrent-native-apo-test PRIVATE ole32 uuid)
  add_test(NAME native-apo COMMAND soundcurrent-native-apo-test $<TARGET_FILE:soundcurrent-native-apo>)
endif()
if(MSVC)
  foreach(target soundcurrent-native-processor soundcurrent-native-processor-test soundcurrent-native-apo soundcurrent-native-apo-test soundcurrent-native-endpoint-audit)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  endforeach()
else()
  target_compile_options(soundcurrent-native-processor PRIVATE -Wall -Wextra -Wpedantic)
endif()
